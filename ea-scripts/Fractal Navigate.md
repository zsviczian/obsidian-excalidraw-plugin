/*
╭──────────────────────────────────────────────────────────────────────────────╮
│ FRACTAL NAVIGATE v1.0.0                                                      │
│ The interaction layer that makes Fractal Index maps feel fractal.           │
│                                                                              │
│ WHAT IT DOES                                                                │
│   Registers a link-click hook for the plugin's lifetime:                     │
│     • click a POD (label, frame, or embedded preview)                        │
│         → the canvas dives into that pod — animated zoom to fit the          │
│           embedded sub-diagram, so you descend INTO the fractal              │
│           without leaving the drawing                                        │
│     • click the same pod again (or any empty area) while diving              │
│         → zoom back out to the whole map                                     │
│     • FILE cards and breadcrumbs keep their native behavior                  │
│         (open the note / climb to the parent index)                          │
│     • Ctrl/Cmd+click always keeps native behavior (escape hatch)             │
│                                                                              │
│ INSTALL (pick one)                                                           │
│   A. Lifetime (recommended): Settings → Excalidraw → Scripts →               │
│      "Startup script" → point it at this file. The hook is active from       │
│      app start until shutdown.                                               │
│   B. Session: run this script once from any drawing (Scripts panel); the     │
│      hook stays active until Obsidian restarts.                              │
│                                                                              │
│ Only acts on elements generated by Fractal Index (customData.fractalIndex).  │
╰──────────────────────────────────────────────────────────────────────────────╯
*/

if (!ea.verifyMinimumPluginVersion || !ea.verifyMinimumPluginVersion("2.0.0")) {
  new Notice("Fractal Index: this script requires Excalidraw plugin version 2.0.0 or newer. Please update the Excalidraw plugin.");
  return;
}


/* per-view dive state: view → key of the embed we zoomed into */
const diveState = new WeakMap();

/* decide what a click on an element means (pure — unit tested) */
function decideAction(el) {
  const cd = el && el.customData;
  if (!cd || cd.fractalIndex !== true) return "native";
  const kind = cd.kind;
  if (kind === "pod-label" || kind === "embed" || kind === "frame" || kind === "dive-hint" || kind === "pod-motif") return "dive";
  return "native"; // file cards, titles, breadcrumbs, arrows → native
}

/* from a clicked pod element's key ("pod|<path>|label"), build the embed key */
function embedKeyOf(el) {
  const key = el.customData.key || "";
  const parts = key.split("|");
  if (parts.length < 3) return null;
  const path = parts.slice(1, -1).join("|");
  return "pod|" + path + "|embed";
}

function findElement(view, key) {
  try {
    const els = view.getViewElements ? view.getViewElements() : [];
    return els.find((el) => el.customData && el.customData.key === key && !el.isDeleted) || null;
  } catch (e) {
    return null;
  }
}

function zoomToEmbed(view, el) {
  // defer past the click handler's synchronous aftermath, or the zoom
  // gets overridden; the plugin's own zoom-to-element (K-Plex API) is the dive
  setTimeout(() => {
    try {
      if (typeof view.zoomToElementId === "function") {
        view.zoomToElementId(el.id, false).catch((e) => console.error("dive failed", e));
        return;
      }
      const api = view.excalidrawAPI || (view.getExcalidrawAPI ? view.getExcalidrawAPI() : null);
      if (api && typeof api.scrollToContent === "function") {
        api.scrollToContent(el, { fitToViewport: true, viewportZoomFactor: 0.9 });
      } else if (view.zoomToFit) {
        view.zoomToFit(false);
      }
    } catch (e) {
      console.error("Fractal Navigate dive error:", e);
    }
  }, 60);
}

function surface(view) {
  // deferred like the dive — a synchronous zoom inside the click handler
  // gets overridden by the click's aftermath
  setTimeout(() => {
    try {
      if (view.zoomToFit) view.zoomToFit(false);
    } catch (e) {
      console.error("Fractal Navigate surface error:", e);
    }
  }, 60);
  diveState.delete(view);
}

/* the hook — registered on the plugin-global EA when reachable, so a manual
   run re-registers the persistent hook; falls back to the session EA */
const hookTarget = (ea.plugin && ea.plugin.ea) ? ea.plugin.ea : ea;
hookTarget.onLinkClickHook = (element, linkText, event, view, eaInstance) => {
  try {
    if (event && (event.ctrlKey || event.metaKey)) return true; // escape hatch
    const action = decideAction(element);
    if (action !== "dive") return true;
    if (!view) return true;

    const embedKey = embedKeyOf(element);
    if (!embedKey) return true;

    if (diveState.get(view) === embedKey) {
      surface(view); // second click on the pod we're inside → zoom back out
      return false;
    }
    const embedEl = findElement(view, embedKey);
    if (!embedEl) return true; // sub-index not generated yet → native open
    zoomToEmbed(view, embedEl);
    diveState.set(view, embedKey);
    return false; // prevent native link open — we handled it
  } catch (e) {
    console.error("Fractal Navigate error:", e);
    return true; // never break native behavior on error
  }
};


/* ══════════════════════════════════════════════════════════════════════════
   LIVE ARROW TRACKING — a binding engine on top of onSceneChangeHook.
   The Obsidian Excalidraw fork ships binding *fields* but no binding
   *engine* (updateBoundElements etc. are absent), so moved cards would
   orphan their arrows. This tracker re-anchors Fractal Index arrows
   (note links, spine stubs, spine) whenever their endpoint cards move.
   It converges: re-anchoring only writes when endpoints actually change.
   ══════════════════════════════════════════════════════════════════════════ */

const trackTimers = new WeakMap(); // view → debounce timer

function rectOf(el, pad) {
  return { x: el.x - pad, y: el.y - pad, w: el.width + 2 * pad, h: el.height + 2 * pad };
}

function anchorPoint(r, toward) {
  const cx = r.x + r.w / 2, cy = r.y + r.h / 2;
  const dx = toward.x + toward.w / 2 - cx, dy = toward.y + toward.h / 2 - cy;
  if (!dx && !dy) return [cx, cy];
  const sx = dx ? (r.w / 2) / Math.abs(dx) : Infinity;
  const sy = dy ? (r.h / 2) / Math.abs(dy) : Infinity;
  const s = Math.min(sx, sy);
  return [cx + dx * s, cy + dy * s];
}

function endpointRect(elements, endpoint) {
  // endpoint is a file path or a pod pseudo-key "folder/"
  if (endpoint.endsWith("/")) {
    const frame = elements.find(
      (el) => el.customData && el.customData.key === "pod|" + endpoint.slice(0, -1) + "|frame" && !el.isDeleted
    );
    return frame ? { x: frame.x, y: frame.y, w: frame.width, h: frame.height } : null;
  }
  const card = elements.find(
    (el) => el.customData && el.customData.key === "file|" + endpoint && !el.isDeleted
  );
  return card ? rectOf(card, 10) : null;
}

function idealLinkPoints(elements, arrow) {
  const spec = String((arrow.customData || {}).key || "").replace(/^link\|/, "");
  const sep = spec.indexOf("=>");
  if (sep < 0) return null;
  const ra = endpointRect(elements, spec.slice(0, sep));
  const rb = endpointRect(elements, spec.slice(sep + 2));
  if (!ra || !rb) return null;
  return [anchorPoint(ra, rb), anchorPoint(rb, ra)];
}

function idealStubPoints(elements, arrow) {
  const key = String((arrow.customData || {}).key || "");
  if (!key.startsWith("pod|") || !key.endsWith("|arrow")) return null;
  const label = elements.find((el) => el.customData && el.customData.key === key.replace(/\|arrow$/, "|label") && !el.isDeleted);
  if (!label) return null;
  return [[-40, label.y + 18], [label.x - 4, label.y + 18]];
}

function idealSpinePoints(elements) {
  const labels = elements.filter(
    (el) => el.customData && el.customData.kind === "pod-label" && !el.isDeleted
  );
  if (!labels.length) return null;
  const lastY = Math.max(...labels.map((l) => l.y + 18));
  return [[-40, 40], [-40, lastY]];
}

function samePoints(a, b) {
  if (!a || !b || a.length !== b.length) return false;
  for (let i = 0; i < a.length; i++) {
    if (Math.abs(a[i][0] - b[i][0]) > 0.5 || Math.abs(a[i][1] - b[i][1]) > 0.5) return false;
  }
  return true;
}

function reanchorNow(elements, appState, view) {
  try {
    // The hook receives the plugin-global EA (no targetView), so we write
    // through the view's imperative API instead of EA plumbing.
    const api = view && view.excalidrawAPI;
    if (!api || typeof api.updateScene !== "function") return;
    const isFractal = elements.some((el) => el.customData && el.customData.fractalIndex);
    if (!isFractal) return;
    const selected = (appState && appState.selectedElementIds) || {};
    const updates = new Map(); // id → ideal absolute points
    for (const el of elements) {
      if (el.isDeleted || !el.customData || !el.customData.fractalIndex) continue;
      if (el.customData.kind !== "link" && el.customData.kind !== "arrow" && el.customData.kind !== "spine") continue;
      if (selected[el.id]) continue; // user is dragging this arrow — hands off
      const ideal =
        el.customData.kind === "link" ? idealLinkPoints(elements, el) :
        el.customData.kind === "arrow" ? idealStubPoints(elements, el) :
        idealSpinePoints(elements);
      const current = (el.points || []).map((p) => [p[0] + el.x, p[1] + el.y]);
      if (!ideal || samePoints(current, ideal)) continue;
      updates.set(el.id, ideal);
    }
    if (!updates.size) return;
    api.updateScene({
      elements: elements.map((el) => {
        const ideal = updates.get(el.id);
        if (!ideal) return el;
        const x0 = ideal[0][0], y0 = ideal[0][1];
        return {
          ...el,
          x: x0,
          y: y0,
          points: ideal.map((p) => [p[0] - x0, p[1] - y0]),
          width: Math.abs(ideal[ideal.length - 1][0] - x0),
          height: Math.abs(ideal[ideal.length - 1][1] - y0),
          elbowed: el.elbowed !== undefined ? el.elbowed : undefined, // preserve orthogonal routing
          version: (el.version || 1) + 1,
        };
      }),
    });
    // programmatic updateScene writes are invisible to the plugin's scene hash
    // and do NOT mark the view dirty — persist the re-anchoring explicitly
    setTimeout(() => {
      try {
        if (typeof view.setDirty === "function") view.setDirty();
        else if (view.forceSave) void view.forceSave(true, false);
      } catch (e) { /* never break tracking over persistence */ }
    }, 80);
  } catch (e) {
    console.error("Fractal Navigate reanchor error:", e);
  }
}

hookTarget.onSceneChangeHook = {
  trackElements: true,
  callback: (elements, appState, files, view, ea) => {
    clearTimeout(trackTimers.get(view));
    trackTimers.set(
      view,
      setTimeout(() => {
        trackTimers.delete(view);
        reanchorNow(elements, appState, view);
      }, 120)
    );
  },
};
new Notice("Fractal Navigate active — click a pod to dive, click again to surface.");
